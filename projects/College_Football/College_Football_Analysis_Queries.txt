To compare Penn State and Wisconsin roster characteristics by team and position:


mysql> SELECT team_name, position, COUNT(*) AS player_count, ROUND(AVG(height_inches), 1) AS avg_height_inches, ROUND(AVG(weight_lbs), 1) AS avg_weight_lbs,  MAX(height_inches) AS tallest_player_inches,  MIN(weight_lbs) AS lightest_player_lbs, MAX(weight_lbs) AS heaviest_player_lbs FROM ( SELECT  'Penn State Nittany Lions' AS team_name, position,  height_inches,  weight_lbs  FROM penn_state_nittany_lions UNION ALL SELECT  'Wisconsin Badgers' AS team_name, position,  height_inches,  weight_lbs FROM wisconsin_badgers) AS combined_rosters GROUP BY team_name,  position ORDER BY position, avg_weight_lbs DESC, team_name;
